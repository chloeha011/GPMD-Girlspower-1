def print_section(title):

    print("\n" + "=" * 65)

# 1. STACK

print_section("1. STACK: LAST IN, FIRST OUT (LIFO)")

print(
    "A stack stores items so that the last item added is the first "
    "item removed. Imagine a pile of plates: you normally take the "
    "top plate first."
)

stack = []

print("\nWorked example:")
print("Start with an empty stack:", stack)

stack.append(10)
print("After append(10):", stack)

stack.append(20)
print("After append(20):", stack)

stack.append(30)
print("After append(30):", stack)

print("The top item is stack[-1]:", stack[-1])

removed_item = stack.pop()
print("pop() removes:", removed_item)
print("Stack after pop():", stack)

print(
    "\nHow it works: append() adds an item to the top of the stack. "
    "pop() removes the most recently added item. Both operations are "
    "usually O(1), meaning their running time does not grow with the "
    "number of stack items."
)

print(
    "\nReal-life examples: Undo in a text editor, browser backtracking, "
    "and the function-call stack used by a programming language."
)


# 2. QUEUE

print_section("2. QUEUE: FIRST IN, FIRST OUT (FIFO)")

print(
    "A queue processes items in the order they arrive. Imagine people "
    "waiting in a line: the first person to arrive is normally served first."
)

queue = deque()

print("\nWorked example:")
print("Start with an empty queue:", list(queue))

queue.append("Alice")
print("Alice joins:", list(queue))

queue.append("Bob")
print("Bob joins:", list(queue))

queue.append("Charlie")
print("Charlie joins:", list(queue))

served_person = queue.popleft()
print("popleft() serves:", served_person)
print("Queue after serving:", list(queue))

queue.append("David")
print("David joins at the back:", list(queue))

print(
    "\nHow it works: append() adds an item to the back of the queue. "
    "popleft() removes the item at the front. With collections.deque, "
    "these operations are usually O(1)."
)

print(
    "\nReal-life examples: Customer service lines, printer jobs, and "
    "tasks waiting to be processed."
)


# 3. LINKED LIST

print_section("3. LINKED LIST: CONNECTED NODES")

print(
    "A linked list is made of nodes. Each node stores a value and a "
    "reference to the next node. The last node points to None, which "
    "marks the end of the list."
)


class Node:
    """One node in a singly linked list."""

    def __init__(self, data):
        self.data = data
        self.next = None


def display_linked_list(head):
    """Print every value by following the links from the head node."""
    current = head
    values = []

    while current is not None:
        values.append(str(current.data))
        current = current.next

    print(" -> ".join(values) + " -> None")


print("\nWorked example:")
first = Node(10)
second = Node(20)
third = Node(30)

first.next = second
second.next = third

print("Original list:")
display_linked_list(first)

print("\nInsert a new node with value 5 at the beginning:")
new_node = Node(5)
new_node.next = first
first = new_node
display_linked_list(first)

print(
    "\nHow it works: Each node points to the next node. To visit the "
    "whole list, start at the head and follow next until reaching None. "
    "Inserting at the beginning is O(1). Finding a particular value "
    "usually takes O(n), because nodes may need to be visited one by one."
)

print(
    "\nReal-life analogy: Think of train carriages connected one after "
    "another. Each carriage is linked to the next one."
)
