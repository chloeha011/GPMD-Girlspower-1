from collections import deque


def show(title):
    print(f"\n{'=' * 45}")
    print(title)
    print("=" * 45)


# 1. STACK — LIFO
show("1. STACK: Last In, First Out")

stack = [10, 20, 30]
print("Initial stack:", stack)

stack.append(40)
print("Push 40:", stack)

print("Pop:", stack.pop())
print("Final stack:", stack)
print("Time complexity: Push O(1), Pop O(1)")
print("Real-life example: Undo in a text editor")


# 2. QUEUE — FIFO
show("2. QUEUE: First In, First Out")

queue = deque(["Alice", "Bob", "Charlie"])
print("Initial queue:", list(queue))

queue.append("David")
print("David joins:", list(queue))

print("Served:", queue.popleft())
print("Remaining queue:", list(queue))
print("Time complexity: Enqueue O(1), Dequeue O(1)")
print("Real-life example: Customer service line")


# 3. LINKED LIST
show("3. LINKED LIST: Connected Nodes")


class Node:
    def __init__(self, data):
        self.data = data
        self.next = None


def display(head):
    current = head
    values = []

    while current:
        values.append(str(current.data))
        current = current.next

    print(" -> ".join(values) + " -> None")


head = Node(10)
head.next = Node(20)
head.next.next = Node(30)

print("Original list:")
display(head)

new_node = Node(5)
new_node.next = head
head = new_node

print("After inserting 5:")
display(head)
print("Insert at beginning: O(1)")
print("Search for a value: O(n)")
print("Real-life analogy: Connected train carriages")

print("Queue      | FIFO | Add: append | Remove: popleft")
print("Linked List| Nodes connected through references")
